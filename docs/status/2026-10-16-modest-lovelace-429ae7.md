Причина: приложение в тесте не слушало порт. Тогда supertest на каждый из ~130 запросов делает `listen(0)`, то есть
занимает `::` (все адреса), и стучится на `127.0.0.1:порт`. macOS разрешает другому процессу занять явным номером
`127.0.0.1:тот же порт` поверх нашего `::`, и более точный адрес выигрывает: запрос брони получает чужой сервер.
Проверено на этой машине: два сервера с портом 0 не сталкиваются никогда (20 000 попыток, 0 совпадений; 6000 чужих
слушателей заранее, тест зелёный), сталкивается только явный номер порта, как у серверов, поднятых на «найденном
свободном порте».

Исправление: в `beforeAll` `await app.listen(0, '127.0.0.1')` вместо `app.init()`; свой адрес занять уже нельзя,
порт один на весь файл. Сторож в том же файле: «приложение само слушает 127.0.0.1».

```js
// sweeper.cjs: чужой процесс занимает 127.0.0.1:P явным номером по кругу и отвечает 405
const http = require('http');
const FIRST = 49152, LAST = 65535, K = 2000, HOLD = 200;
let p = FIRST, live = 0;
function bindNext() {
  if (live >= K) return;
  const port = p; p = p === LAST ? FIRST : p + 1; live++;
  const s = http.createServer((q, r) => { r.statusCode = 405; r.end(); });
  s.on('error', () => { live--; setImmediate(bindNext); });
  s.listen(port, '127.0.0.1', () => {
    setTimeout(() => s.close(() => { live--; bindNext(); }), HOLD);
    setImmediate(bindNext);
  });
}
for (let i = 0; i < 50; i++) bindNext();
```
