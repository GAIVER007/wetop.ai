import { test, expect } from '@playwright/test';
const qa = 'http://127.0.0.1:55825';
test.beforeEach(async ({ page, request }) => {
  const f = await (await request.post(`${qa}/__qa/reset`, { data: {} })).json();
  await page.goto('/auth/fallback');
  await page.getByRole('main').getByLabel('Email', { exact: true }).fill(f.email);
  await page.getByRole('main').getByLabel('Пароль', { exact: true }).fill(f.password);
  await page.getByRole('button', { name: 'Войти', exact: true }).click();
  await page.getByRole('button', { name: 'Продолжить', exact: true }).click();
  await expect(page.getByRole('main').getByLabel('Название бизнеса')).toBeVisible();
});
test.afterAll(async ({ request }) => expect((await request.post(`${qa}/__qa/cleanup`)).ok()).toBe(true));
for (const mode of ['offline', 'timeout', 'before503', 'after']) {
  test(`RED a5 ${mode}: must keep form and offer safe retry`, async ({ page, request }) => {
    await page.getByRole('main').getByLabel('Название бизнеса').fill(`QA-RED-${mode}`);
    await request.post(`${qa}/__qa/fault`, { data: { mode } });
    const terminal = page.waitForResponse(r => new URL(r.url()).pathname === '/register/setup' && r.request().method() === 'POST');
    await page.getByRole('button', { name: 'Сохранить', exact: true }).click();
    await (await terminal).finished();
    await expect(page.getByRole('main').getByLabel('Название бизнеса')).toHaveValue(`QA-RED-${mode}`);
    await expect(page.getByRole('button', { name: 'Повторить', exact: true })).toBeVisible();
  });
}
test('RED a5 lost next must have explicit safe repeat', async ({ page, request }) => {
  await request.post(`${qa}/__qa/fault`, { data: { mode: 'after' } });
  const terminal = page.waitForResponse(r => new URL(r.url()).pathname === '/register/setup' && r.request().method() === 'POST');
  await page.getByRole('button', { name: 'Продолжить', exact: true }).click();
  await (await terminal).finished();
  await expect(page.getByRole('button', { name: 'Повторить', exact: true })).toBeVisible();
});
test('RED a5 conflict must keep input and offer explicit recovery', async ({ page, context }) => {
  const other = await context.newPage();
  await other.goto('/register/setup');
  await other.getByRole('main').getByLabel('Название бизнеса').fill('QA-RED-other');
  await other.getByRole('button', { name: 'Сохранить', exact: true }).click();
  await expect(other.getByRole('status')).toHaveText('Сохранено');
  await page.getByRole('main').getByLabel('Название бизнеса').fill('QA-RED-local');
  await page.getByRole('button', { name: 'Сохранить', exact: true }).click();
  await expect(page.getByRole('main').getByLabel('Название бизнеса')).toHaveValue('QA-RED-local');
  await expect(page.getByRole('button', { name: 'Загрузить сохранённую версию' })).toBeVisible();
});
